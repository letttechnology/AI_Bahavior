```java
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.openai.OpenAiChatModel;
import org.springframework.ai.vertexai.gemini.VertexAiGeminiChatModel;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.stereotype.Service;

@Service
public class MultiModelChatService {

    private final ChatClient geminiClient;
    private final ChatClient openAiClient;

    // Spring AI automatically creates these ChatModel beans from your starters
    public MultiModelChatService(VertexAiGeminiChatModel geminiModel, OpenAiChatModel openAiModel) {
        // Build dedicated clients for each model
        this.geminiClient = ChatClient.builder(geminiModel).build();
        this.openAiClient = ChatClient.builder(openAiModel).build();
    }

    public String generateResponse(String userPrompt) {
        try {
            // Try using Google Gemini first
            return this.geminiClient.prompt()
                    .user(userPrompt)
                    .call()
                    .content();
        } catch (Exception e) {
            // If Gemini is rate-limited or blocking, seamlessly fall back to OpenAI
            System.err.println("Gemini failed or rate-limited. Falling back to OpenAI: " + e.getMessage());
            
            return this.openAiClient.prompt()
                    .user(userPrompt)
                    .call()
                    .content();
        }
    }
}
```

import org.springframework.ai.chat.model.ChatModel;

public class ManagedModel {
    private final String providerName;
    private final ChatModel chatModel;
    private boolean monthlyQuotaExceeded = false;

    public ManagedModel(String providerName, ChatModel chatModel) {
        this.providerName = providerName;
        this.chatModel = chatModel;
    }

    // Getters and Setters
    public String getProviderName() { return providerName; }
    public ChatModel getChatModel() { return chatModel; }
    public boolean isMonthlyQuotaExceeded() { return monthlyQuotaExceeded; }
    public void setMonthlyQuotaExceeded(boolean monthlyQuotaExceeded) { 
        this.monthlyQuotaExceeded = monthlyQuotaExceeded; 
    }

import org.springframework.ai.chat.client.ChatClient;
import org.springframework.stereotype.Component;
import java.util.List;
import java.util.concurrent.atomic.AtomicInteger;

@Component
public class ResilientAiRouter {

    private final List<ManagedModel> modelPool;
    private final AtomicInteger nextModelIndex = new AtomicInteger(0);

    // Spring passes the configured pool directly from our upcoming Config class
    public ResilientAiRouter(List<ManagedModel> modelPool) {
        this.modelPool = modelPool;
    }

    public String executeWithFailover(String userPrompt) {
        int poolSize = modelPool.size();
        int attempts = 0;

        // Loop up to the total number of providers to find a working option
        while (attempts < poolSize) {
            int targetIndex = nextModelIndex.getAndIncrement() % poolSize;
            ManagedModel managedModel = modelPool.get(targetIndex);

            // 1. Skip completely if marked as monthly rate/budget exceeded
            if (managedModel.isMonthlyQuotaExceeded()) {
                attempts++;
                continue;
            }

            try {
                // 2. Wrap the chosen model in a ChatClient on the fly and call it
                return ChatClient.builder(managedModel.getChatModel())
                        .build()
                        .prompt(userPrompt)
                        .call()
                        .content();

            } catch (Exception ex) {
                System.err.printf("Provider [%s] failed or rate-limited. Trying next... Reason: %s%n", 
                        managedModel.getProviderName(), ex.getMessage());

                // 3. Inspect the error to see if it's a permanent monthly budget cap issue
                if (isMonthlyQuotaError(ex)) {
                    managedModel.setMonthlyQuotaExceeded(true);
                    System.err.printf("CRITICAL: Provider [%s] deactivated for the month due to billing caps.%n", 
                            managedModel.getProviderName());
                }

                attempts++;
            }
        }

        throw new IllegalStateException("All configured AI providers are currently exhausted or rate-limited!");
    }

    /**
     * Inspects error responses for common indicators that your monthly billing/quota has rolled over.
     */
    private boolean isMonthlyQuotaError(Exception ex) {
        String errMsg = ex.getMessage().toLowerCase();
        return errMsg.contains("quota") || errMsg.contains("budget") || errMsg.contains("billing") || errMsg.contains("credit");
    }
}

The core limitation is that Spring Boot’s auto-configuration only processes **one** `spring.ai.openai` properties block. If you try to duplicate that block in your `application.yaml`, the properties will simply overwrite each other.

     ```java
             List<ManagedModel> available = models.stream().filter(m -> !m.isMonthlyQuotaExceeded()).toList();
             if (available.isEmpty()) throw new IllegalStateException("[" + tier + "-tier] No AI providers available");
             int startIdx = Math.abs(counter.getAndIncrement() % available.size());
             for (int i = 0; i < available.size(); i++) {

                 try {

                     ChatClient.ChatClientRequestSpec spec = client.prompt();
                     if (systemPrompt != null && !systemPrompt.isBlank()) spec = spec.system(systemPrompt);
     ```
